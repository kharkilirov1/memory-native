from ana3 import parse, ms, X, OLD
import re
Q = X + r'\q\\'
def P(n): return parse(Q + n + '.log')
def mean(v): return st.mean(v)
def sd(v): return st.stdev(v) if len(v) > 1 else float('nan')
def f(x, n=4): return f"{x:.{n}f}"
def sgn(x, n=4): return f"{x:+.{n}f}"
out = []
def o(s=''): out.append(s)

arms = ['dense_nobias', 'qat_fp4', 'tern_counter', 'na_fp4_c8', 'lg6_fp4_h2']
ds = [128, 256, 512]
D = {}
for a in arms:
    for d in ds:
        D[(a, d, 'base')] = P(f'{a}_d{d}')
for a in ['dense_nobias', 'qat_fp4', 'lg6_fp4_h2']:
    D[(a, 512, 'a1e-3')] = P(f'{a}_d512_a1e-3')
D[('dense_nobias', 512, 'a5e-4')] = P('dense_nobias_d512_a5e-4')

def st_(rows, key='val'):
    v = [r[key] for r in rows]; return mean(v), sd(v), len(v)
def num(rows, key):
    vals = []
    for r in rows:
        try: vals.append(float(r[key]))
        except Exception: pass
    return mean(vals) if vals else float('nan')

    flag = ' DIVERGED' if (t == 'base' and d == 512 and a in ('dense_nobias', 'qat_fp4')) else ''
    bits = rows[0]['bits']

        mr, sr, _ = st_(rr); md, sdd, _ = st_(ref[d])
        g.append((mr - md, math.sqrt(sr ** 2 + sdd ** 2)))

    o(f"| {a} | {sgn(g[0][0])} | {sgn(g[1][0])} | {sgn(g[2][0])} | {sgn(g[2][0]-g[0][0])} | {note} |")
